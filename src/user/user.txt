import configparser
import os
import sys
from pathlib import Path
from functions import functions as f

config = configparser.ConfigParser()

config.read(Path(__file__).resolve().parent / "settings" / "settings.ini")
if config["main"]["userdone"] == "true":
    inp = input(
        "User already finished setup, if this is false, enter continue to continue"
    )
    if inp != "continue":
        raise SystemExit


user = f.open_packages("user", None)
f.pacman_install(user)
print("All user packages have been installed")

try:
    flats = f.open_packages("user", "flathub")
    f.flatpak_install(flats)
except Exception as e:
    f.exit_from_error(e)
else:
    print("All Flatpak packages have been installed")
try:
    aur = f.open_packages("user", "aur")
    f.yay_install(aur)
except Exception as e:
    f.exit_from_error(e)
else:
    print("All AUR packages have been installed")

)
try:
    with open(
